### 3. upload_to_r2.js

**功能**: 上传视频到Cloudflare R2

**使用AWS SDK**:
```javascript
const { S3Client, PutObjectCommand } = require('@aws-sdk/client-s3');

const r2Client = new S3Client({
    region: 'auto',
    endpoint: config.R2_ENDPOINT,
    credentials: {
        accessKeyId: config.R2_ACCESS_KEY_ID,
        secretAccessKey: config.R2_SECRET_ACCESS_KEY
    }
});

// 上传文件
await r2Client.send(new PutObjectCommand({
    Bucket: config.R2_BUCKET_NAME,
    Key: fileName,
    Body: fileContent,
    ContentType: 'video/mp4'
}));

// 返回公开URL
return `${config.R2_PUBLIC_URL}/${fileName}`;
```

### 创建排期帖子

**Endpoint**: `POST /api/v2/scheduler/posts`

**Payload结构**:
```javascript
{
    text: "文案内容",
    media: ["视频URL"],
    publicationDate: {
        dateTime: "2025-11-12T20:00:00",  // UTC+0格式
        timezone: "Asia/Shanghai"
    },
    providers: [
        { network: "instagram" },
        { network: "tiktok" },
        { network: "threads" },
        { network: "facebook" },
        { network: "youtube" }
    ],
    instagramData: { type: "REEL" },
    tiktokData: {
        disableComment: false,
        disableDuet: false,
        disableStitch: false,
        privacyOption: "PUBLIC_TO_EVERYONE"
    },
    threadsData: {},
    facebookData: { type: "REEL" },
    youtubeData: {
        title: "视频标题",
        type: "short",
        privacy: "public",
        madeForKids: false
    },
    autoPublish: true,
    draft: false
}
```

### 获取已排期帖子

**Endpoint**: `GET /api/v2/scheduler/posts`

**参数**:
- `start`: 开始时间 (ISO 8601)
- `end`: 结束时间 (ISO 8601)
- `timezone`: 时区
- `userId`: 用户ID
- `blogId`: 博客ID

### 删除帖子

**Endpoint**: `DELETE /api/v2/scheduler/posts/{postId}`

**参数**:
- `postId`: 帖子ID
- `userId`: 用户ID
- `blogId`: 博客ID

## 🔄 工作流程

### 发布流程

```
1. 读取 Excel → 获取贴文数据
2. 遍历每条记录：
   a. 检查状态 → 跳过"已排期"
   b. 查找本地视频 → 精确/模糊匹配
   c. 上传到R2 → 获取公开URL
   d. 调用Metricool API → 创建排期
   e. 更新Excel → 记录URL和时间
3. 保存Excel → 完成
```

### 排期时间计算

```javascript
// 每天晚上8点，一天一条
const baseTime = new Date();
baseTime.setDate(baseTime.getDate() + index);  // 第index天
baseTime.setHours(20, 0, 0, 0);  // 晚上8点

// 转换为UTC（Metricool需要）
const shanghaiTime = new Date(baseTime.getTime() + (8 * 60 * 60 * 1000));
const dateTimeStr = shanghaiTime.toISOString().split('.')[0];
```

## 🛠️ 自定义修改

### 修改排期策略

**位置**: `process_excel_posts.js` 第188-193行

**示例1：每天两次（10点和16点）**
```javascript
const dayOffset = Math.floor(index / 2);  // 两个帖子一天
const timeSlot = index % 2;  // 0=上午, 1=下午
baseTime.setDate(baseTime.getDate() + dayOffset);
baseTime.setHours(timeSlot === 0 ? 10 : 16, 0, 0, 0);
```

**示例2：工作日发布（跳过周末）**
```javascript
let daysToAdd = 0;
let postsScheduled = 0;

while (postsScheduled < index + 1) {
    const checkDate = new Date();
    checkDate.setDate(checkDate.getDate() + daysToAdd);

    // 跳过周六(6)和周日(0)
    if (checkDate.getDay() !== 0 && checkDate.getDay() !== 6) {
        postsScheduled++;
    }
    daysToAdd++;
}

baseTime.setDate(baseTime.getDate() + daysToAdd - 1);
baseTime.setHours(20, 0, 0, 0);
```

### 添加重试机制

系统已内置axios-retry：

```javascript
axiosRetry(metricoolAxios, {
    retries: 5,  // 重试5次
    retryDelay: axiosRetry.exponentialDelay,  // 指数退避
    retryCondition: (error) => {
        return axiosRetry.isNetworkOrIdempotentRequestError(error)
            || error.code === 'ECONNABORTED'
            || error.code === 'ETIMEDOUT'
            || (error.response && error.response.status >= 500);
    }
});
```

## 📊 错误处理

### 错误类型

1. **视频文件不存在**
   ```
   ❌ 未找到本地视频文件: {title}
   ```
   → 检查文件名是否匹配

2. **R2上传失败**
   ```
   ❌ 上传R2失败: {error}
   ```
   → 检查R2配置和网络

3. **Metricool API错误**
   ```
   ❌ Metricool上传失败: {error}
   状态码: {status}
   ```
   → 检查Token和参数

4. **TikTok Token过期**
   ```
   Error 40131: Refresh token is invalid or expired
   ```
   → 在Metricool中重新授权
